[auth]
auth		authenticate with linkhash
auth signup	create a new linkhash account
auth login	store username/token for deka commands
auth logout	clear saved credentials
auth whoami	show current authenticated user
signup		alias for `deka auth signup`
login		alias for `deka auth login`
logout		alias for `deka auth logout`
whoami		alias for `deka auth whoami`

[database]
db		database tooling for DekaScript ORM generation and migrations
db migrate	apply pending db migrations
db info	show db generation and migration state
db flush	reset database schema (dev only)

[debug]
introspect		inspect deka-runtime scheduler state
introspect top	show top isolates
introspect workers	show worker stats
introspect inspect	inspect a specific isolate
introspect kill	kill a specific isolate

[extensions]
wasm		manage wasm extensions
wasm init	scaffold a wasm extension
wasm build	build a wasm extension
wasm stubs	generate wasm .d.ds stubs

[internal]
self		internal deka maintenance commands
self fetch	fetch a pinned content checkout (testsuite, tour)
self monitor	run the long-running self-update daemon
self test	run internal deka compatibility and content tests
self update	update deka components

[package]

unlink		remove a local package link without touching its target
pkg		package operations
pkg install	install package(s)
pkg link	link a local package working tree
pkg update	update dependencies to latest within semver range
pkg unlink	remove a local package link
pkg publish	publish package
pkg release	bump, tag, push, and publish package
publish		publish a DekaScript package release to Linkhash
release		bump, tag, push, and publish a package release

[pipeline]
deploy		linkhash pipeline deployment and execution
deploy run	run a pipeline from a linkhash.yaml file in a gild VM

[project]
init		initialize a new app project

compile		compile to single-file executable
transpile		emit JavaScript from a .ds file or directory (execs dsc)
verify		verify dist/ against the artifact manifest's payload digests

[runtime]

run		run the app

serve		serve a handler or directory
task		run a task from deka.json
test		run DekaScript tests
dev		serve with HTTP + HMR (dev mode)
