[getting started]
init		scaffold a new project
serve		run it as a local dev server
dev		same, plus hot reload while you edit
add		add a package to your project
build		build your project for production
run		run a single .ds/.dsx file or script

[auth]
auth		authenticate with linkhash
auth signup	create a new linkhash account
auth login	store username/token for deka commands
auth logout	clear saved credentials
auth whoami	show current authenticated user
signup		alias for `deka auth signup`
login		alias for `deka auth login`
logout		alias for `deka auth logout`
whoami		alias for `deka auth whoami`

[database]
db		database tooling for DekaScript ORM generation and migrations
db migrate	apply pending db migrations
db info	show db generation and migration state
db flush	reset database schema (dev only)

[debug]
introspect		inspect deka-runtime scheduler state
introspect top	show top isolates
introspect workers	show worker stats
introspect inspect	inspect a specific isolate
introspect kill	kill a specific isolate

[extensions]
wasm		manage wasm extensions
wasm init	scaffold a wasm extension
wasm build	build a wasm extension
wasm stubs	generate wasm .d.ds stubs

[internal]
self		internal deka maintenance commands
self fetch	fetch a pinned content checkout (testsuite, tour)
self test	run internal deka compatibility and content tests

[package]
install		install dependencies via the package manager
add		install package(s) from the index
i		install package(s) from the index
update		update dependencies to latest within semver range
summon		fetch, vet and lock a JavaScript URL or JSR package
link		link a local package working tree into this project
unlink		remove a local package link without touching its target
pkg		package operations
pkg install	install package(s)
pkg link	link a local package working tree
pkg update	update dependencies to latest within semver range
pkg unlink	remove a local package link
pkg publish	publish package
pkg release	bump, tag, push, and publish package
publish		publish a DekaScript package release to Linkhash
release		bump, tag, push, and publish a package release

[pipeline]
deploy		linkhash pipeline deployment and execution
deploy run	run a pipeline from a linkhash.yaml file in a gild VM

[project]
init		initialize a new app project
build		build a DekaScript file into a JavaScript module
check		validate a DekaScript file or the whole project (execs dsc)
fmt		format DekaScript source or emitted JavaScript (execs dsc)
compile		compile to single-file executable
transpile		emit JavaScript from a .ds file or directory (execs dsc)
verify		verify dist/ against the artifact manifest's payload digests

[runtime]
cache		manage the user-global cache
cache clear	remove all loose-file cache entries
run		run the app
platform		multi-tenant platform server — serves per-tenant DekaScript handlers from tenants/ directory
serve		serve a handler or directory
task		run a task from deka.json
test		run DekaScript tests
dev		serve with HTTP + HMR (dev mode)

[flags]
--help		show help
--version		show version
--verbose		show detailed metadata where supported
--debug		enable debug logging
--allow-read		allow filesystem reads (`--allow-read=./src,./data`)
--allow-write		allow filesystem writes
--allow-net		compatibility flag; serve/run/platform read net policy from deka.json
--allow-env		compatibility flag; serve/run/platform read env policy from deka.json
--allow-run		allow subprocess execution
--allow-db		allow database operations
--allow-dynamic		allow dynamic code execution
--allow-wasm		allow wasm module load and execution
--allow-all		allow all security capabilities
--deny-read		deny filesystem reads (`--deny-read=/etc`)
--deny-write		deny filesystem writes
--deny-net		compatibility flag; serve/run/platform read net policy from deka.json
--deny-env		compatibility flag; serve/run/platform read env policy from deka.json
--deny-run		deny subprocess execution
--deny-db		deny database operations
--deny-dynamic		deny dynamic code execution
--deny-wasm		deny wasm module load and execution
--no-prompt		disable interactive security prompts
